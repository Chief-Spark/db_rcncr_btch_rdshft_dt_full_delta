   
	(IN	 P_Lote			INTEGER

	,IN	 P_StringSQL		VARCHAR(32000)
	,IN	 P_Descripcion	VARCHAR(1000)
	,OUT P_ERROR		VARCHAR(32000)

/*
 * ################################################################################
 * 
  * Objetivo:		Actualiza el registro de la ejecucion en la tabla LOG_TD_DETALLE
 * Fecha:		2016/09/27
 * Realizado por:	Luis E Ramirez
 * Parametros:		P_Lote  --> Lote de la ejecuci??n de la tabla LOG_TD
 * 			P_StartTimeStampPadre --> Fecha de inicio del Lote
 *			P_Nombre_Proceso_Padre --> Nombre del proceso de la LOG_TD
 *			P_StringSQL --> Query
 *			P_Descripcion --> Alguna descripcion
 *			P_ERROR	--> Error salida.
 * Proyecto:		Reconocer 3
 * Script:		SP_EXECSTRING_DES.sql
 * 
 * 
 * 
 * BITACORA
 * ---------------------------------------------------------------------------
 * Fecha		Objetivo			Realizado por
 *

 *

 * 
 *  Este componente fue realizado bajo la metodologia de desarrollo de Capgemini S.A.S y
 * 		se encuentra protegido por las leyes de derechos de autor
 * 
 * 
 * ################################################################################
 */

BEGIN
	

	------------------------
	-- Declaracion de variables
	------------------------
	DECLARE		V_Log_TD_Detalle_ID	INTEGER;
	DECLARE 	V_ERROR			INTEGER;
	DECLARE 	V_SQLCODE		INTEGER;
	DECLARE 	V_DESC_ERR		VARCHAR(500);
	DECLARE		V_MENSAJE		VARCHAR(1000);
	DECLARE 	V_STATUS		VARCHAR(1000);
	
	---------------------------
	-- Inicializacion de variables
	---------------------------
	SET P_ERROR = '';
	SET V_Log_TD_Detalle_ID = 0;

	--------------------
	-- Se Busca el numero del paso
	------------------
L1:	BEGIN
	
	--------------------
	-- Define manejo de error
	--------------------

	DECLARE EXIT HANDLER FOR SQLEXCEPTION  
	BEGIN 
		SET V_STATUS = 'Error al ' || P_Descripcion;
    	SET V_SQLCODE = SQLCODE;
    		
		--------------------
		-- Define descripcion de error
		--------------------
	        
		SELECT	ErrorText INTO V_DESC_ERR
		FROM	dbc.ErrorMsgs
		WHERE 	ERRORCODE = SQLCODE;
    	
    		SET V_MENSAJE = 'SQLCODE ' || V_SQLCODE || ' : ' || V_DESC_ERR;
    		
		CALL REC_DATOS.SP_UPDLOGTDDETALLE_DES (:P_Lote
											   ,V_Log_TD_Detalle_ID
											   ,V_MENSAJE
											   ,V_STATUS
											   );
					    
		SET P_ERROR = V_MENSAJE;
	END;

  
	--------------------
	-- Inicia registro en LOG_TD_DETALLE
	--------------------
	CALL REC_DATOS.SP_INSLOGTDDETALLE_DES(:P_Lote
										 ,:P_StartTimeStampPadre
										 ,:P_Nombre_Proceso_Padre
										 ,:P_StringSQL
										 ,V_Log_TD_Detalle_ID);

	--------------------
	-- Ejecuta cadena
	--------------------				
    	CALL DBC.SysExecSQL(:P_StringSQL);
    	    	
	--------------------
	-- Actualiza LOG_TD_DETALLE
	--------------------
    	SET V_MENSAJE = 'Finalizo Exitosamente';
    	SET V_STATUS = V_MENSAJE || ' al ' || P_Descripcion;
    	
    	CALL REC_DATOS.SP_UPDLOGTDDETALLE_DES (:P_Lote
												,V_Log_TD_Detalle_ID
												,V_MENSAJE
												,V_STATUS
											);
    
	END L1;
    
END;
